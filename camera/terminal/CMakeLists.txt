cmake_minimum_required(VERSION 3.20)
project(vision_terminal LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(_FILE_OFFSET_BITS=64)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(VISION_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/build/deps" CACHE PATH "Prepared, version-pinned dependency source directory")
foreach(header json/single_include/nlohmann/json.hpp httplib/httplib.h lvgl/lvgl.h freetype/include/ft2build.h)
  if(NOT EXISTS "${VISION_DEPS}/${header}")
    message(FATAL_ERROR "Missing ${header}; run prepare_deps.ps1")
  endif()
endforeach()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_graph.py" "${CMAKE_CURRENT_BINARY_DIR}/model_graph.h" COMMAND_ERROR_IS_FATAL ANY)
add_library(vision_core STATIC core.cpp level_compat.cpp)
target_include_directories(vision_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(vision-engine engine.cpp hardware.cpp ../int8/gray.cpp)
target_include_directories(vision-engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${VISION_DEPS}/json/single_include" "${VISION_DEPS}/httplib")
target_link_libraries(vision-engine PRIVATE vision_core Threads::Threads)
foreach(dependency ZLIB BZIP2 PNG HARFBUZZ BROTLI)
  set(FT_DISABLE_${dependency} ON CACHE BOOL "" FORCE)
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${VISION_DEPS}/freetype" freetype EXCLUDE_FROM_ALL)
set(LV_BUILD_CONF_PATH "" CACHE PATH "" FORCE)
set(LV_BUILD_CONF_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${VISION_DEPS}/lvgl" lvgl EXCLUDE_FROM_ALL)
target_link_libraries(lvgl PUBLIC freetype)
add_executable(vision-panel panel.cpp management.cpp xp_style.cpp xp_assets.cpp)
target_include_directories(vision-panel PRIVATE "${VISION_DEPS}/json/single_include" "${VISION_DEPS}/httplib")
target_link_libraries(vision-panel PRIVATE lvgl freetype Threads::Threads m)
install(TARGETS vision-engine vision-panel RUNTIME DESTINATION bin)
install(FILES vision_auth.py vision_mqtt.py DESTINATION bin)
install(DIRECTORY "${VISION_DEPS}/python/paho" DESTINATION bin)
install(PROGRAMS manager.py DESTINATION bin RENAME vision-manager.py)
install(FILES linux/vision-engine.service linux/vision-panel.service linux/vision-manager.service DESTINATION /etc/systemd/system)
